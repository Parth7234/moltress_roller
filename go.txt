------DIJKSTRA-------

#include <iostream>
#include <vector>
#include <queue>
#include <climits> 

using namespace std;

vector<int> dijkstra(vector<vector<pair<int,int>>> &g,vector<int> &dist){
    priority_queue<pair<int,int>,vector<pair<int,int>>,greater<pair<int,int>>> pq;
    pq.push({0,1});
    vector<int> parent(dist.size());
    while(!pq.empty()){
        int u=pq.top().second;
        int d=pq.top().first;
        pq.pop();
        if(d>dist[u])continue;
        for(auto v:g[u]){
            if(dist[v.second]>dist[u]+v.first){
                dist[v.second]=dist[u]+v.first;
                parent[v.second]=u;
                pq.push({dist[v.second],v.second});
            }
        }
    }
    return parent;
}

int main(){
    int n,m;
    cin>>n>>m;
    vector<vector<pair<int,int>>> g(n+1);
    for(int i=0;i<m;i++){
        int u,v,w;
        cin>>u>>v>>w;
        g[u].push_back({w,v});
        g[v].push_back({w,u});
    }
    vector<int> dist(n+1,INT_MAX);
    dist[1]=0;
    vector<int> parent=dijkstra(g,dist);
    if(dist[n]==INT_MAX)cout<<-1<<endl;
    else{
        vector<int> s;
        int cur=n;
        s.push_back(n);
        while(cur!=1){
            s.push_back(parent[cur]);
            cur=parent[cur];
        }
        for(int i=s.size()-1;i>=0;i--){
            cout<<s[i]<<" ";
        }
        cout<<endl;
    }
}

-------FLOYD WARSHALL---------
#include<iostream> 
using namespace std;

void floydWarshall(vector<vector<int>> &dist,int v,vector<vector<int>> &p){
    for(int k=0;k<v;k++){
        for(int i=0;i<v;i++){
            for(int j=0;j<v;j++){
                if(dist[i][j]>dist[i][k]+dist[k][j]){
                    p[i][j]=p[k][j];
                }
                dist[i][j]=min(dist[i][j],dist[i][k]+dist[k][j]);
            }
        }
    }
}

void print(vector<vector<int>> &p,int i,int j){
    if(i==j)cout<<i<<endl;
    else if(p[i][j]==-1)cout<<"No path is possible\n";
    else{
        print(p,i,p[i][j]);
        cout<<j<<endl;
    }
}

int main(){
    int INF = 100000000;
    vector<vector<int>> dist = {
        {0, 4, INF, 5, INF},
        {INF, 0, 1, INF, 6},
        {2, INF, 0, 3, INF},
        {INF, INF, 1, 0, 2},
        {1, INF, INF, 4, 0}
    };
    vector<vector<int>> p(5,vector<int>(5));
    for(int i=0;i<5;i++){
        for(int j=0;j<5;j++){
            if(dist[i][j]==INF||i==j)p[i][j]=-1;
            else p[i][j]=i;
        }
    }
    floydWarshall(dist,5,p);
    for(int i = 0; i<dist.size(); i++) {
        for(int j = 0; j<dist.size(); j++) {
            cout<<dist[i][j]<<" ";
        }
        cout<<endl;
    }
    print(p,1,3);
}

----------KNAPSACK------

#include<iostream>

using namespace std;

signed main(){
    int n;
    cin>>n;
    int W;
    cin>>W;
    vector<int> v(n);
    for(int i=0;i<n;i++){
        cin>>v[i];
    }
    vector<int> w(n);
    for(int i=0;i<n;i++){
        cin>>w[i];
    }
    vector<pair<double,int>> k(n);
    for(int i=0;i<n;i++){
        k[i]={(double)v[i]/w[i],i};
    }
    sort(k.rbegin(),k.rend());
    int rem=W;
    double ans=0;
    for(int i=0;i<n;i++){
        int ind=k[i].second;
        int val=v[ind];
        int weight=w[ind];
        if(rem>weight){
            rem-=weight;
            ans+=val;
        }
        else{
            ans+=(rem)*((double)val/weight);
            rem=0;
            break;
        }
    }
    cout<<ans<<endl;
}

----------LCS---------
#include<iostream>

using namespace std;

int main(){
    string a,b;
    cin>>a>>b;
    int n=a.size();
    int m=b.size();
    vector<vector<int>> dp(n+1,vector<int>(m+1,0));
    for(int i=0;i<n;i++){
        for(int j=0;j<m;j++){
            if(a[i]==b[j]){
                dp[i+1][j+1]=dp[i][j]+1;
            }
            else{
                dp[i+1][j+1]=max(dp[i][j+1],dp[i+1][j]);
            }
        }
    }
    int i=n;
    int j=m;
    string ans;
    while(i>0&&j>0){
        if(a[i-1]==b[j-1]){
            ans=a[i-1]+ans;
            i--;
            j--;
        }
        else if(dp[i-1][j]>dp[i][j-1]){
            i--;
        }
        else j--;
    }
    cout<<ans<<endl;
}

#include<iostream>

using namespace std;

void primMST(vector<vector<int>> &graph){
    int v=graph.size();
    vector<int> visited(v,0);
    vector<int> key(v,INT_MAX);
    vector<int> parent(v);
    key[0]=0;
    parent[0]=-1;
    for(int i=0;i<v-1;i++){
        int min=INT_MAX;
        int u=-1;
        for(int j=0;j<v;j++){
            if(!visited[j]&&key[j]<min){
                min=key[j];
                u=j;
            }
        }
        if(u==-1)break;
        visited[u]=1;
        for(int j=0;j<v;j++){
            if(graph[u][j]&&!visited[j]&&key[j]>graph[u][j]){
                key[j]=graph[u][j];
                parent[j]=u;
            }
        }
    }
    int ans=0;
    for(int i=1;i<v;i++){//dont include key (here key was 0 so it was not included)
        ans+=graph[i][parent[i]];
        cout<<parent[i]<<" "<<i<<" "<<graph[i][parent[i]];
        cout<<endl;
    }
    cout<<ans<<endl;
    cout<<endl;
}

int main(){
    vector<vector<int>> graph = { { 0, 2, 0, 6, 0 },
                        		{ 2, 0, 3, 8, 5 },
                        		{ 0, 3, 0, 0, 7 },
                        		{ 6, 8, 0, 0, 9 },
                        		{ 0, 5, 7, 9, 0 } };

    primMST(graph);
}

#include<iostream>

#define int long long
using namespace std;

signed main(){
    string s;
    cin>>s;
    string p;
    cin>>p;
    int n=s.size();
    int m=p.size();
    int pr=31;
    vector<int> prime(m+1);
    prime[0]=1;
    int M=1e9+9;
    for(int i=1;i<=m;i++){
        prime[i]=((prime[i-1]%M)*pr)%M;
    }
    int cur=0;
    int val=0;
    vector<int> ans;
    for(int i=0;i<m;i++){
        cur+=(((s[i]-'a'+1)%M)*(prime[m-i-1]%M));
        val+=(((p[i]-'a'+1)%M)*(prime[m-i-1]%M));
    }
    for(int i=0;i<=n-m;i++){
        if(cur==val){
            bool flag=true;
            for(int j=i;j<i+m;j++){
                if(s[j]!=p[j-i]){
                    flag=false;
                    break;
                }
            }
            if(flag)ans.push_back(i);
        }
        if(i==n-m)break;
        int l=((s[i]-'a'+1)*prime[m-1])%M;
        cur=(cur-l+M)%M;
        cur=(cur%M*pr)%M;
        cur+=(((s[i+m]-'a'+1)%M))%M;
    }
    if(ans.size()==0)cout<<-1<<endl;
    else{
        for(int i=0;i<ans.size();i++){
            cout<<ans[i]<<" ";
        }
        cout<<endl;
    }
}

---------TASK-SCHEDULING------------

#include<iostream> 
#include <queue>
using namespace std;

int main(){
    int n;
    cin>>n;
    vector<pair<int,int>> v(n);
    for(int i=0;i<n;i++){
        cin>>v[i].first;
        cin>>v[i].second;
    }
    sort(v.begin(),v.end());
    priority_queue<int,vector<int>,greater<int>> pq;
    for(int i=0;i<n;i++){
        pair<int,int> cur=v[i];
        if(cur.first<=pq.size()&&cur.second>pq.top()){
            pq.pop();
            pq.push(cur.second);
        }
        else if(cur.first>pq.size()){
            pq.push(cur.second);
        }
    }
    int ct=0;
    int ans=0;
    while(!pq.empty()){
        ct++;
        ans=ans+(pq.top());
        pq.pop();
    }
    cout<<ct<<" "<<ans<<endl;
}
